size_t pkt_decoded_length(const uint8_t *src, size_t len, int escaped)
{
    size_t need = 0;
    size_t i = 0;

    while (i < len) {
        uint8_t c = src[i];
        if (escaped && c == '\\') {
            if (i + 1 >= len) {
                need += 1;
                i += 1;
                continue;
            }
            need += 1;
            i += 2;
            continue;
        }

        need += 1;
        i += 1;
    }

    return need;
